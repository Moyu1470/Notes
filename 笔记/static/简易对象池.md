```C#
using UnityEngine;  
using System.Collections.Generic;  
  
public class SimplePool<T> where T : MonoBehaviour  
{  
    private readonly T _prefab;  //声明T类型字段 来存储预设体
    private readonly Stack<T> _stack = new Stack<T>();  //创建堆栈（栈） 来存储预制体对象
  
    public SimplePool(T prefab, int initial = 0)  //创建简易对象池的构造函数 （需要输入对象类型T 和对象池大小）
    {        
        _prefab = prefab;  //保存预设体类别
        for (int i = 0; i < initial; i++)//遍历对象池  
        {            
            var obj = GameObject.Instantiate(prefab); //实例化对象  
            obj.gameObject.SetActive(false);  //将实例化出来的对象失活
            _stack.Push(obj);  //将对象压入对象池
        }   
    }  
    public T Get()  
    {        
        if (_stack.Count > 0)  //如果对象池中仍有对象
        {            
            var o = _stack.Pop();  //直接弹出并接收记录对象
            o.gameObject.SetActive(true);  //将对象设置为激活状态
            return o;  
        }        
        return GameObject.Instantiate(_prefab);   //如果对象池中没有空闲对象了 就新建预制体
    }  
    public void Release(T obj)  
    {        
        obj.gameObject.SetActive(false);  //
        _stack.Push(obj);  
    }
}
```
